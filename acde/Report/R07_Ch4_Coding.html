4.3 Critical Code — Deployment Pipeline (orchestrator.js)

async function runDeploymentPipeline({ project, deployment }) { const log = (msg) => appendLog(deployment._id, msg); // Step 1 — Clone const repoDir = path.join(config.workspaceDir, String(project._id), 'src'); await cloneRepo(project.git_repo_url, repoDir, log); // Step 2 — Detect stack & generate Dockerfile if absent const stack = detectStack(repoDir); ensureDockerfile(repoDir, stack); // Step 3 — Build image const image = imageTag(project._id); await buildImage(repoDir, image, log); // Step 4 — Deploy (Docker or K8s) const deployer = config.deployMode === 'k8s' ? deployK8s : deployDocker; const result = await deployer({ project, deployment, image, log }); // Step 5 — Proxy route + public tunnel await upsertRoute(deployment.subdomain, deployment.assigned_port); const tunnelUrl = await openTunnel(deployment.subdomain, deployment.assigned_port); deployment.live_url = tunnelUrl || liveUrl(deployment.subdomain, deployment.assigned_port); deployment.status = 'deployed'; await deployment.save(); }

4.3 Critical Code — OS-Level Port Probe (ports.js)

// Probes OS TCP stack to guarantee the port is genuinely free function isPortFree(port) { return new Promise(resolve => { const srv = net.createServer(); srv.once('error', () => resolve(false)); // EADDRINUSE srv.once('listening', () => srv.close(() => resolve(true))); srv.listen(port, '0.0.0.0'); }); } // allocatePort() calls isPortFree() and skips busy ports

This fix was applied after a Docker bind error where port 31000 was held by a previous container not recorded in MongoDB. The OS probe catches such orphaned bindings.
